cmake_minimum_required(VERSION 3.22)
project(V21Presence LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(PRESENCE_ASAN "Standalone tests only; never execute mangosd" OFF)
set(GAME ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/game)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(target presence_test presence_benchmark)
 add_executable(${target} ${target}.cpp)
 if(target STREQUAL "presence_test")
  target_sources(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/presence_ops.cpp)
  target_compile_definitions(${target} PRIVATE PRESENCE_COUNT)
 else()
  target_sources(${target} PRIVATE ${GAME}/PlayerActivityPresence.cpp)
 endif()
 target_include_directories(${target} PRIVATE ${GAME} ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_link_options(${target} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 if(PRESENCE_ASAN)
  target_compile_options(${target} PRIVATE /fsanitize=address)
 endif()
endforeach()
add_test(NAME presence_lifecycle_parity_operations COMMAND presence_test)
set_tests_properties(presence_lifecycle_parity_operations PROPERTIES TIMEOUT 90)
if(PRESENCE_ASAN)
 get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(presence_lifecycle_parity_operations PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
