cmake_minimum_required(VERSION 3.22)
project(V23Attackers LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(V23_ASAN "AddressSanitizer" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark allocations)
 if(V23_ASAN AND NOT mode STREQUAL "test")
  continue()
 endif()
 add_executable(attackers_${mode} main.cpp)
 target_compile_features(attackers_${mode} PRIVATE cxx_std_17)
 target_compile_options(attackers_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_compile_definitions(attackers_${mode} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX MANGOSBOT_ZERO)
 target_link_options(attackers_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 target_include_directories(attackers_${mode} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
 if(mode STREQUAL "test")
  target_compile_definitions(attackers_${mode} PRIVATE V23_TRACE)
 endif()
 if(mode STREQUAL "allocations")
  target_sources(attackers_${mode} PRIVATE ../../v22_hotpaths/allocations.cpp)
  target_compile_definitions(attackers_${mode} PRIVATE V23_ALLOCATIONS)
 endif()
 if(V23_ASAN)
  target_compile_options(attackers_${mode} PRIVATE /fsanitize=address)
 endif()
endforeach()
add_test(NAME v23_attackers_parity COMMAND attackers_test)
set_tests_properties(v23_attackers_parity PROPERTIES TIMEOUT 180)
if(V23_ASAN)
 set_tests_properties(v23_attackers_parity PROPERTIES TIMEOUT 600)
 get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(v23_attackers_parity PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
