cmake_minimum_required(VERSION 3.22)
project(V23LogAction LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(V23_ASAN "AddressSanitizer" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark allocations)
 if(V23_ASAN AND NOT mode STREQUAL "test")
  continue()
 endif()
 add_executable(log_action_${mode} main.cpp)
 target_compile_features(log_action_${mode} PRIVATE cxx_std_17)
 target_compile_options(log_action_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_compile_definitions(log_action_${mode} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
 target_link_options(log_action_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 target_include_directories(log_action_${mode} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
 if(mode STREQUAL "test")
  target_compile_definitions(log_action_${mode} PRIVATE V23_TRACE)
 endif()
 if(mode STREQUAL "allocations")
  target_sources(log_action_${mode} PRIVATE ../../v22_hotpaths/allocations.cpp)
  target_compile_definitions(log_action_${mode} PRIVATE V23_ALLOCATIONS)
 endif()
 if(V23_ASAN)
  target_compile_options(log_action_${mode} PRIVATE /fsanitize=address)
 endif()
endforeach()
add_test(NAME v23_log_action_parity COMMAND log_action_test)
set_tests_properties(v23_log_action_parity PROPERTIES TIMEOUT 180)
if(V23_ASAN)
 get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(v23_log_action_parity PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
