cmake_minimum_required(VERSION 3.22)
project(V20CheckValuesAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(CHECKVALUES_ASAN "AddressSanitizer for extracted CheckValues contracts" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
set(targets checkvalues_audit)
if(NOT CHECKVALUES_ASAN)
    list(APPEND targets checkvalues_benchmark)
endif()
foreach(target IN LISTS targets)
    add_executable(${target} main.cpp consumer.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    if(target STREQUAL "checkvalues_audit")
        target_compile_definitions(${target} PRIVATE CHECKVALUES_COUNT_ALLOC)
    endif()
    if(MSVC)
        file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/${target})
        target_compile_options(${target} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG /FAcs /Fa${CMAKE_CURRENT_BINARY_DIR}/${target}/)
        target_link_options(${target} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
        if(CHECKVALUES_ASAN)
            target_compile_options(${target} PRIVATE /fsanitize=address)
        endif()
    endif()
endforeach()
add_test(NAME checkvalues_contracts COMMAND checkvalues_audit)
if(CHECKVALUES_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(checkvalues_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
