cmake_minimum_required(VERSION 3.22)
project(PresenceDiagnosticsAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(PRESENCE_ASAN "ASan for standalone tests only (never execute mangosd)" OFF)
set(BOTS ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/game/PlayerBots/playerbot)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(target presence_test presence_benchmark)
    add_executable(${target} ${target}.cpp ${BOTS}/PresenceDiagnostics.cpp)
    target_include_directories(${target} PRIVATE ${BOTS} ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
    if(PRESENCE_ASAN)
        target_compile_options(${target} PRIVATE /fsanitize=address)
    endif()
endforeach()
target_compile_definitions(presence_test PRIVATE PRESENCE_TEST_CLOCK)
add_test(NAME presence_parity_and_collector COMMAND presence_test)
set_tests_properties(presence_parity_and_collector PROPERTIES TIMEOUT 60)
if(PRESENCE_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(presence_parity_and_collector PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
