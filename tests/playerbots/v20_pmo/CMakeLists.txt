cmake_minimum_required(VERSION 3.22)
project(V20PmoReportAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(V20_PMO_ASAN "AddressSanitizer for actual PrintStats contracts" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py
    ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_executable(v20_pmo_test main.cpp)
target_include_directories(v20_pmo_test PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(v20_pmo_test PRIVATE cxx_std_17)
target_compile_options(v20_pmo_test PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
target_link_options(v20_pmo_test PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
if(V20_PMO_ASAN)
    target_compile_options(v20_pmo_test PRIVATE /fsanitize=address)
endif()
add_test(NAME v20_pmo_report_contracts COMMAND v20_pmo_test)
set_tests_properties(v20_pmo_report_contracts PROPERTIES TIMEOUT 60)
if(V20_PMO_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(v20_pmo_report_contracts PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
