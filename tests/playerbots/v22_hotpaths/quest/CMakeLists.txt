cmake_minimum_required(VERSION 3.22)
project(V22QuestHotPath LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(V22_ASAN "AddressSanitizer" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark allocations)
 if(V22_ASAN AND NOT mode STREQUAL "test")
  continue()
 endif()
 add_executable(quest_${mode} main.cpp services.cpp)
 add_executable(chain_${mode} grind_chain.cpp services.cpp)
 target_compile_features(chain_${mode} PRIVATE cxx_std_17)
 target_compile_options(chain_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_link_options(chain_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 target_include_directories(chain_${mode} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_features(quest_${mode} PRIVATE cxx_std_17)
 target_compile_options(quest_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_link_options(quest_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 if(mode STREQUAL "allocations")
  target_sources(quest_${mode} PRIVATE ../allocations.cpp)
  target_sources(chain_${mode} PRIVATE ../allocations.cpp)
  target_compile_definitions(quest_${mode} PRIVATE AUDIT_ALLOCATIONS)
 endif()
 if(mode STREQUAL "test")
  target_compile_definitions(quest_${mode} PRIVATE QUEST_TRACE)
  target_compile_definitions(chain_${mode} PRIVATE QUEST_TRACE)
 endif()
 foreach(variant IN ITEMS v21 v22)
  set(obj ${mode}_${variant})
  add_library(${obj} OBJECT factory.cpp consumer.cpp)
  target_compile_features(${obj} PRIVATE cxx_std_17)
  target_compile_options(${obj} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
  target_include_directories(${obj} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${obj} PRIVATE AUDIT_NAMESPACE=${variant} AUDIT_HEADER="${variant}.h" AUDIT_BODIES="${variant}_bodies.inc" AUDIT_API=${variant}_api AUDIT_LABEL="${variant}" AUDIT_B=1)
  if(mode STREQUAL "test")
   target_compile_definitions(${obj} PRIVATE QUEST_TRACE)
  endif()
  if(V22_ASAN)
   target_compile_options(${obj} PRIVATE /fsanitize=address)
  endif()
  target_link_libraries(quest_${mode} PRIVATE ${obj})
  target_link_libraries(chain_${mode} PRIVATE ${obj})
 endforeach()
 if(V22_ASAN)
  target_compile_options(quest_${mode} PRIVATE /fsanitize=address)
  target_compile_options(chain_${mode} PRIVATE /fsanitize=address)
 endif()
endforeach()
add_test(NAME v22_quest_contracts COMMAND quest_test)
add_test(NAME v22_grind_quest_chain COMMAND chain_test)
set_tests_properties(v22_quest_contracts PROPERTIES TIMEOUT 180)
if(V22_ASAN)
 get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(v22_quest_contracts v22_grind_quest_chain PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
