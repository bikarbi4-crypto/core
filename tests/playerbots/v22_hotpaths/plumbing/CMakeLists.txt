cmake_minimum_required(VERSION 3.22)
project(V22Plumbing LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(V22_ASAN "AddressSanitizer" OFF)
option(V22_MEMORY_MONITOR "Retain memory monitor copy semantics" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark allocations)
 if(V22_ASAN AND NOT mode STREQUAL "test")
  continue()
 endif()
 add_executable(plumbing_${mode} main.cpp)
 target_compile_features(plumbing_${mode} PRIVATE cxx_std_17)
 target_compile_options(plumbing_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
 target_link_options(plumbing_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
 if(V22_MEMORY_MONITOR)
  target_compile_definitions(plumbing_${mode} PRIVATE MEMORY_MONITOR)
 endif()
 if(mode STREQUAL "allocations")
  target_sources(plumbing_${mode} PRIVATE ../allocations.cpp)
 endif()
 foreach(variant IN ITEMS v21 v22)
  set(obj ${mode}_${variant})
  add_library(${obj} OBJECT consumer.cpp)
  target_compile_features(${obj} PRIVATE cxx_std_17)
  target_compile_options(${obj} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
  target_include_directories(${obj} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${obj} PRIVATE MANGOSBOT_ZERO AUDIT_NAMESPACE=${variant} AUDIT_HEADER="${variant}.h" AUDIT_API=${variant}_api AUDIT_LABEL="${variant}")
  if(mode STREQUAL "test")
   target_compile_definitions(${obj} PRIVATE PLUMBING_TRACE)
  endif()
  if(V22_MEMORY_MONITOR)
   target_compile_definitions(${obj} PRIVATE MEMORY_MONITOR)
  endif()
  if(V22_ASAN)
   target_compile_options(${obj} PRIVATE /fsanitize=address)
  endif()
  target_link_libraries(plumbing_${mode} PRIVATE ${obj})
 endforeach()
 if(V22_ASAN)
  target_compile_options(plumbing_${mode} PRIVATE /fsanitize=address)
 endif()
endforeach()
add_test(NAME v22_plumbing_contracts COMMAND plumbing_test)
if(V22_ASAN)
 get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(v22_plumbing_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
