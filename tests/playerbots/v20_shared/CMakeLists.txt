cmake_minimum_required(VERSION 3.22)
project(V20SharedLifetime LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(SHARED_ASAN "Instrument actual constructor/lifetime path" OFF)
option(SHARED_ALLOCATIONS "Separate allocation accounting executable; never time this build" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
file(GLOB GAME_DIRS LIST_DIRECTORIES true "${ROOT}/src/game/*")
set(INCLUDES ${ROOT}/src/game ${ROOT}/src/framework ${ROOT}/src/shared ${ROOT}/dep/include
 ${ROOT}/src/game/PlayerBots ${ROOT}/src/game/PlayerBots/playerbot ${ROOT}/src/game/Maps/Pool
 ${ROOT}/src/game/Movement/spline ${ROOT}/src/game/Server/Protocol
 ${ROOT}/dep/recastnavigation ${ROOT}/dep/recastnavigation/Detour
 ${ROOT}/dep/g3dlite/include ${ROOT}/dep/windows/include ${ROOT}/dep/windows/include/mysql
 ${ROOT}/dep/windows/include/openssl ${ROOT}/dep/windows/include/zlib
 ${ROOT}/dep/windows/optional_dependencies/curl/include ${CMAKE_CURRENT_BINARY_DIR}/generated)
foreach(D IN LISTS GAME_DIRS)
 if(IS_DIRECTORY "${D}")
  list(APPEND INCLUDES "${D}")
 endif()
endforeach()
foreach(V baseline candidate)
 add_executable(shared_${V} main.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/constructors.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/monitor.cpp)
 target_compile_features(shared_${V} PRIVATE cxx_std_17)
 target_include_directories(shared_${V} PRIVATE ${INCLUDES})
 target_compile_definitions(shared_${V} PRIVATE ENABLE_PLAYERBOTS DO_MYSQL _WIN32 WIN32 _WINDOWS OPENSSL_NO_DEPRECATED SHARED_HEADER="${V}.h" SHARED_VARIANT="${V}")
 if(SHARED_ALLOCATIONS)
  if(SHARED_ASAN)
   message(FATAL_ERROR "Use separate ASan and allocation builds")
  endif()
  target_sources(shared_${V} PRIVATE allocations.cpp)
  target_compile_definitions(shared_${V} PRIVATE SHARED_ALLOCATIONS)
 endif()
 if(V STREQUAL "candidate")
  target_compile_definitions(shared_${V} PRIVATE SHARED_CANDIDATE)
 endif()
 if(MSVC)
  target_compile_options(shared_${V} PRIVATE /GR /GL- /MP1 /W3 /bigobj /Gy /FI${CMAKE_CURRENT_SOURCE_DIR}/fixture_prefix.h)
  target_link_options(shared_${V} PRIVATE /INCREMENTAL:NO /OPT:REF /OPT:NOICF /LTCG:OFF)
  if(SHARED_ASAN)
   target_compile_options(shared_${V} PRIVATE /fsanitize=address /Zi)
  endif()
 endif()
endforeach()
enable_testing()
add_test(NAME shared_candidate COMMAND shared_candidate --check)
add_test(NAME shared_baseline_off_contracts COMMAND shared_baseline --check)
if(MSVC AND SHARED_ASAN)
 add_test(NAME shared_v19_expected_uaf COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/verify_uaf.py $<TARGET_FILE:shared_baseline> ${CMAKE_CURRENT_BINARY_DIR}/v19-expected-uaf.log)
 get_filename_component(MSVC_RUNTIME "${CMAKE_CXX_COMPILER}" DIRECTORY)
 set_tests_properties(shared_candidate shared_baseline_off_contracts shared_v19_expected_uaf PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${MSVC_RUNTIME}")
endif()
