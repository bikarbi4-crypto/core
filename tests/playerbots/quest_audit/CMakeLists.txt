cmake_minimum_required(VERSION 3.22)
project(PlayerbotQuestAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(QUEST_ASAN "AddressSanitizer for production quest contracts" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark)
    if(QUEST_ASAN AND mode STREQUAL "benchmark")
        continue()
    endif()
    add_executable(quest_${mode} main.cpp services.cpp)
    target_compile_features(quest_${mode} PRIVATE cxx_std_17)
    target_compile_options(quest_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
    target_link_options(quest_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
    if(mode STREQUAL "test")
        target_compile_definitions(quest_${mode} PRIVATE QUEST_TRACE)
    endif()
    foreach(variant IN ITEMS v171 a_only b_only c_only v18)
        set(obj ${mode}_${variant})
        add_library(${obj} OBJECT factory.cpp consumer.cpp)
        target_compile_features(${obj} PRIVATE cxx_std_17)
        target_compile_options(${obj} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
        target_include_directories(${obj} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
        if(variant STREQUAL "b_only" OR variant STREQUAL "v18")
            set(b 1)
        else()
            set(b 0)
        endif()
        target_compile_definitions(${obj} PRIVATE AUDIT_NAMESPACE=${variant} AUDIT_HEADER="${variant}.h" AUDIT_BODIES="${variant}_bodies.inc" AUDIT_API=${variant}_api AUDIT_LABEL="${variant}" AUDIT_B=${b})
        if(mode STREQUAL "test")
            target_compile_definitions(${obj} PRIVATE QUEST_TRACE)
        endif()
        if(QUEST_ASAN)
            target_compile_options(${obj} PRIVATE /fsanitize=address)
        endif()
        target_link_libraries(quest_${mode} PRIVATE ${obj})
    endforeach()
    if(QUEST_ASAN)
        target_compile_options(quest_${mode} PRIVATE /fsanitize=address)
    endif()
endforeach()
add_test(NAME quest_contracts COMMAND quest_test)
set_tests_properties(quest_contracts PROPERTIES TIMEOUT 120)
if(QUEST_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(quest_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
