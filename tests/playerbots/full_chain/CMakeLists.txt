cmake_minimum_required(VERSION 3.22)
project(PlayerbotFullChainAudit LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(AUDIT_CANDIDATE "Include the working-tree candidate" OFF)
set(args "")
if(AUDIT_CANDIDATE)
    list(APPEND args --candidate)
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated ${args}
    COMMAND_ERROR_IS_FATAL ANY)
set(variants v16 c_only ab_only v17)
if(AUDIT_CANDIDATE)
    list(APPEND variants candidate)
endif()
add_executable(full_chain_audit main.cpp services.cpp)
target_compile_features(full_chain_audit PRIVATE cxx_std_17)
if(AUDIT_CANDIDATE)
    target_compile_definitions(full_chain_audit PRIVATE AUDIT_CANDIDATE)
endif()
foreach(variant IN LISTS variants)
    add_library(${variant}_objects OBJECT factory.cpp consumer.cpp)
    target_compile_features(${variant}_objects PRIVATE cxx_std_17)
    target_include_directories(${variant}_objects PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
    if(variant STREQUAL "v16" OR variant STREQUAL "ab_only")
        set(metadata 0)
    else()
        set(metadata 1)
    endif()
    target_compile_definitions(${variant}_objects PRIVATE AUDIT_NAMESPACE=${variant} AUDIT_HEADER="${variant}.h"
        AUDIT_API=${variant}_api AUDIT_LABEL="${variant}" AUDIT_METADATA=${metadata})
    if(MSVC)
        target_compile_options(${variant}_objects PRIVATE /GR /GL- /MP1 /W3 /FAcs /Fa${CMAKE_CURRENT_BINARY_DIR}/${variant}/)
    endif()
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/${variant})
    target_link_libraries(full_chain_audit PRIVATE ${variant}_objects)
endforeach()
if(MSVC)
    target_compile_options(full_chain_audit PRIVATE /GR /GL- /MP1 /W3)
    target_link_options(full_chain_audit PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
endif()
