cmake_minimum_required(VERSION 3.22)
project(V20TargetsAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TARGETS_ASAN "AddressSanitizer for extracted target filtering" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_executable(targets_audit main.cpp)
target_compile_definitions(targets_audit PRIVATE TARGETS_COUNT_ALLOC)
target_include_directories(targets_audit PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(targets_audit PRIVATE cxx_std_17)
target_compile_options(targets_audit PRIVATE /GR /GL- /MP1 /W4 /UNDEBUG)
target_link_options(targets_audit PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
if(TARGETS_ASAN)
    target_compile_options(targets_audit PRIVATE /fsanitize=address)
else()
    target_compile_options(targets_audit PRIVATE /FAs)
    add_executable(targets_benchmark main.cpp)
    target_include_directories(targets_benchmark PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_compile_features(targets_benchmark PRIVATE cxx_std_17)
    target_compile_options(targets_benchmark PRIVATE /GR /GL- /MP1 /W4 /UNDEBUG /FAs)
    target_link_options(targets_benchmark PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
endif()
add_test(NAME targets_contracts COMMAND targets_audit)
set_tests_properties(targets_contracts PROPERTIES TIMEOUT 180)
if(TARGETS_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(targets_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
