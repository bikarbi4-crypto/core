cmake_minimum_required(VERSION 3.22)
project(PlayerbotGrindAmortizationAudit LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_executable(grind_amortization_audit main.cpp)
target_include_directories(grind_amortization_audit PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(grind_amortization_audit PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(grind_amortization_audit PRIVATE /GR /GL- /MP1 /W4 /WX /UNDEBUG)
    target_link_options(grind_amortization_audit PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
endif()
