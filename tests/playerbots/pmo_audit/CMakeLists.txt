cmake_minimum_required(VERSION 3.22)
project(PlayerbotPmoAudit LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(PMO_ASAN "AddressSanitizer for PMO contracts" OFF)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
foreach(mode IN ITEMS test benchmark)
    if(PMO_ASAN AND mode STREQUAL "benchmark")
        continue()
    endif()
    add_executable(pmo_${mode} main.cpp)
    target_include_directories(pmo_${mode} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_compile_features(pmo_${mode} PRIVATE cxx_std_17)
    target_compile_options(pmo_${mode} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
    target_link_options(pmo_${mode} PRIVATE /INCREMENTAL:NO /OPT:NOICF /LTCG:OFF)
    if(mode STREQUAL "test")
        target_compile_definitions(pmo_${mode} PRIVATE PMO_TRACE)
    endif()
    foreach(variant IN ITEMS v18 v19)
        set(obj ${mode}_${variant})
        add_library(${obj} OBJECT consumer.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/${variant}_monitor.cpp)
        target_compile_features(${obj} PRIVATE cxx_std_17)
        target_compile_options(${obj} PRIVATE /GR /GL- /MP1 /W3 /UNDEBUG)
        target_include_directories(${obj} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(${obj} PRIVATE AUDIT_NAMESPACE=${variant} AUDIT_HEADER="${variant}.h" AUDIT_SCOPES="${variant}_scopes.inc" AUDIT_API=${variant}_api AUDIT_LABEL="${variant}")
        if(mode STREQUAL "test")
            target_compile_definitions(${obj} PRIVATE PMO_TRACE)
        elseif(NOT PMO_ASAN)
            # Optimized source+assembly listing, alongside the object files.
            target_compile_options(${obj} PRIVATE /FAs)
        endif()
        if(PMO_ASAN)
            target_compile_options(${obj} PRIVATE /fsanitize=address)
        endif()
        target_link_libraries(pmo_${mode} PRIVATE ${obj})
    endforeach()
    if(PMO_ASAN)
        target_compile_options(pmo_${mode} PRIVATE /fsanitize=address)
    endif()
endforeach()
add_test(NAME pmo_contracts COMMAND pmo_test)
set_tests_properties(pmo_contracts PROPERTIES TIMEOUT 120)
if(PMO_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(pmo_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
endif()
